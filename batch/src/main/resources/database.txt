\c postgres;
DROP DATABASE cluster_btc;
CREATE DATABASE cluster_btc;
\c cluster_btc;

CREATE TABLE ADDRESS (
	ADDRESS_ID BIGSERIAL PRIMARY KEY,
	ADDRESS_HASH VARCHAR(255) UNIQUE NOT NULL,
	MINER_ADDRESS BOOLEAN NOT NULL,
	ADDRESS_TYPE SMALLINT,
	CLUSTER_ID BIGINT
);

CREATE TABLE SUB_CLUSTER (
	ADDRESS_ID_1 BIGINT REFERENCES ADDRESS(ADDRESS_ID),
	ADDRESS_ID_2 BIGINT REFERENCES ADDRESS(ADDRESS_ID),
	CLUSTER_ID BIGINT NOT NULL,
	LINK_TYPE SMALLINT NOT NULL,
	LINK_VISIBLE_SIZE INT DEFAULT 0,
	PRIMARY KEY(ADDRESS_ID_1,ADDRESS_ID_2)
);

select count(*) from (
    select cluster_id, count(*)
    from (select * from address where miner_address = true) as foo1
    group by cluster_id) as foo;

// Numero di cluster che hanno un certo numero di address
select addresses, count(*) as num_cluster from
(select cluster_id, count(*) as addresses from address group by cluster_id) as foo
group by addresses order by addresses;
